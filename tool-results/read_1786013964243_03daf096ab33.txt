    75→    75→function useCountUp(target: number, duration: number = 2000, startOnView: boolean = true) {
    76→    76→  const [count, setCount] = useState(0);
    77→    77→  const ref = useRef<HTMLDivElement>(null);
    78→    78→  const started = useRef(false);
    79→    79→
    80→    80→  useEffect(() => {
    81→    81→    if (!startOnView) return;
    82→    82→    const el = ref.current;
    83→    83→    if (!el) return;
    84→    84→    const observer = new IntersectionObserver(
    85→    85→      ([entry]) => {
    86→    86→        if (entry.isIntersecting && !started.current) {
    87→    87→          started.current = true;
    88→    88→          const startTime = Date.now();
    89→    89→          const animate = () => {
    90→    90→            const elapsed = Date.now() - startTime;
    91→    91→            const progress = Math.min(elapsed / duration, 1);
    92→    92→            const eased = 1 - Math.pow(1 - progress, 3);
    93→    93→            setCount(Math.round(target * eased));
    94→    94→            if (progress < 1) requestAnimationFrame(animate);
    95→    95→          };
    96→    96→          requestAnimationFrame(animate);
    97→    97→        }
    98→    98→      },
    99→    99→      { threshold: 0.3 }
   100→   100→    );
   101→   101→    observer.observe(el);
   102→   102→    return () => observer.disconnect();
   103→   103→  }, [target, duration, startOnView]);
   104→   104→
   105→   105→  return { count, ref };
   106→   106→}
