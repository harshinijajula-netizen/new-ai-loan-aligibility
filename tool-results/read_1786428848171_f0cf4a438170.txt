    31→    31→function useCountUp(target: number, duration = 2000) {
    32→    32→  const [count, setCount] = useState(0);
    33→    33→  const ref = useRef<HTMLDivElement>(null);
    34→    34→  const started = useRef(false);
    35→    35→  useEffect(() => {
    36→    36→    const el = ref.current;
    37→    37→    if (!el) return;
    38→    38→    const observer = new IntersectionObserver(
    39→    39→      ([entry]) => {
    40→    40→        if (entry.isIntersecting && !started.current) {
    41→    41→          started.current = true;
    42→    42→          const t0 = Date.now();
    43→    43→          const tick = () => {
    44→    44→            const p = Math.min((Date.now() - t0) / duration, 1);
    45→    45→            setCount(Math.round(target * (1 - Math.pow(1 - p, 3))));
    46→    46→            if (p < 1) requestAnimationFrame(tick);
    47→    47→          };
    48→    48→          requestAnimationFrame(tick);
    49→    49→        }
    50→    50→      },
    51→    51→      { threshold: 0.3 },
    52→    52→    );
    53→    53→    observer.observe(el);
    54→    54→    return () => observer.disconnect();
    55→    55→  }, [target, duration]);
    56→    56→  return { count, ref };
    57→    57→}
